 * FlixPeriph em vez da Adafruit_MPU6050. Motivo: o MPU vendido como "6050
 * generico" costuma ser um MPU6500, que a lib da Adafruit nao reconhece.
 *
 * Este arquivo NAO inclui ESP32SensorsAccel.hpp (aquele continua na Adafruit,
 * intocado). O modulo de acelerometro esta definido aqui embaixo, e os modulos
 * de Ambiente e LED sao reaproveitados sem alteracao.
 *
 * O JSON enviado ao servidor Python e' identico (mesmas chaves, mesmas unidades
 * em m/s^2), entao nao precisa mexer no servidor.
 *
 * -----------------------------------------------------------------------------
 * ESCOLHA DO SENSOR: mude apenas a linha do "typedef" mais abaixo.
 *   MPU6500 -> seu sensor fisico
 *   MPU6050 -> Wokwi (a peca wokwi-mpu6050 simula o 6050 original)
 * -----------------------------------------------------------------------------
 * platformio.ini para ESTA versao (nao aplicado no projeto, troque quando usar):
 *
 * [env:esp32]
 * platform = https://github.com/pioarduino/platform-espressif32/releases/download/53.03.13/platform-espressif32.zip
 * framework = arduino
 * board = esp32dev
 * monitor_speed = 115200
 * lib_deps =
 *     adafruit/DHT sensor library @ ^1.4.6
 *     adafruit/Adafruit Unified Sensor @ ^1.1.9
 *     bblanchon/ArduinoJson @ ^7.4.1
 *     okalachev/FlixPeriph @ ^1.10.3
 *     ; as duas abaixo so continuam aqui porque ESP32SensorsAccel.hpp (Adafruit)
 *     ; ainda vive em src/ e o LDF do PlatformIO enxerga o arquivo:
 *     adafruit/Adafruit MPU6050 @ ^2.2.2
 *     adafruit/Adafruit BusIO @ ^1.14.1
 *
 * Obs.: so pode existir UM .ino em src/ na hora de compilar. Renomeie o outro
 * (ex.: iot-aula02-app04.ino.bak) antes de buildar este.

/* *Includes* */
//WiFi
#include <WiFi.h>
#include <WiFiClient.h>
//WebServer
#include <WebServer.h>
//IMU
#include <Wire.h>
#include <FlixPeriph.h>
//ESP32Sensors
#include "ESP32SensorsAmbiente.hpp"
#include "ESP32SensorsLED.hpp"

// Pinos
#define DHT_PIN   23
#define DHT_MODEL DHT11
#define LED_PIN   26
#define SDA_PIN   18
#define SCL_PIN   19
//JSON
#include <Arduino.h>
#include <ArduinoJson.h>

/* Modulo do acelerometro via FlixPeriph
 * (substitui ESP32Sensors::Accel do ESP32SensorsAccel.hpp)
 */
namespace ESP32Sensors {
	namespace Accel {

		// >>> TROQUE AQUI: MPU6500 (fisico) ou MPU6050 (Wokwi) <<<
		typedef MPU6500 MPU;

		MPU mpu(Wire);

		// Substitui o sensors_event_t da Adafruit
		struct ACCEL {
			float x;
			float y;
			float z;
			bool valido;
		};

		void inicializar(uint8_t sda, uint8_t scl) {
			// Inicializa comunicação I2C nos pinos informados
			Wire.begin(sda, scl);
			if (!mpu.begin()) {
				Serial.println("Erro: MPU não encontrado. Verifique as conexões.");
				while (1) delay(10);
			}

			// Sensibilidade do acelerometro:
			// ACCEL_RANGE_2G  = mais sensivel
			// ACCEL_RANGE_4G
			// ACCEL_RANGE_8G
			// ACCEL_RANGE_16G = mede acelerações maiores
			mpu.setAccelRange(MPU::ACCEL_RANGE_16G);

			// Filtro passa-baixa interno. O original usava 21 Hz da Adafruit;
			// no FlixPeriph as opções são por faixa aproximada.
			mpu.setDLPF(MPU::DLPF_50HZ_APPROX);
		}

		ACCEL medirAccel() {
			ACCEL dados;
			dados.x = NAN;
			dados.y = NAN;
			dados.z = NAN;
			dados.valido = false;

			if (!mpu.read()) {
				Serial.println("ERRO: leitura incorreta no sensor MPU.");
				return dados;
			}

			// getAccel devolve em m/s^2, igual ao sensors_event_t da Adafruit
			mpu.getAccel(dados.x, dados.y, dados.z);
			dados.valido = true;

			return dados;
		}

	}
}

//WebServer
const char* ssid = "NorisIoT";
const char* password = "Secure10T";
const char* host = "172.16.10.101"; // IP do servidor Python
const uint16_t port = 5000;

//Dispositivo
const char* device = "ESP32Noris";

void setup() {
  Serial.begin(115200);

  //Inicializa todos os sensores
  ESP32Sensors::Ambiente::inicializar(DHT_PIN, DHT_MODEL);
  ESP32Sensors::Accel::inicializar(SDA_PIN, SCL_PIN);
  ESP32Sensors::LED::inicializar(LED_PIN);

  WiFi.begin(ssid, password);
  Serial.print("Conectando ao WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi conectado!");
}

void loop() {
  //Mede temperatura, umidade e índice de calor
  ESP32Sensors::Ambiente::AMBIENTE salaMaquinas = ESP32Sensors::Ambiente::medirAmbiente();

  //Mede trepidação/movimentação do motor (eixos x, y e z)
  ESP32Sensors::Accel::ACCEL motor = ESP32Sensors::Accel::medirAccel();

  WiFiClient client;
  if (client.connect(host, port)) {

    /* Transmissão dos dados em formato JSON */
    JsonDocument doc;

    doc["device"] = device;
    doc["temp"] = salaMaquinas.temp;
    doc["umid"] = salaMaquinas.umid;
    doc["hic"] = salaMaquinas.ic;
    doc["motor_accel_x"] = motor.x;
    doc["motor_accel_y"] = motor.y;
    doc["motor_accel_z"] = motor.z;

    String output;
    serializeJson(doc, output);
    client.println(output);
    Serial.println("Dados enviados: " + output);
    client.stop();
  } else {
    Serial.println("Falha ao conectar no servidor");
  }

  delay(5000); // Aguarda 5s
}
